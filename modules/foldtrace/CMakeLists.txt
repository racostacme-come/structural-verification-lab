cmake_minimum_required(VERSION 3.20)
project(foldtrace VERSION 0.1.0 LANGUAGES CXX)
option(FOLDTRACE_PYTHON "Build Python bindings" ON)
include(CTest)
add_library(continuation INTERFACE)
target_include_directories(continuation INTERFACE cpp)
target_compile_features(continuation INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(continuation INTERFACE /W4 /WX)
else()
  target_compile_options(continuation INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(FOLDTRACE_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/bindings.cpp)
  target_link_libraries(_core PRIVATE continuation)
  install(TARGETS _core LIBRARY DESTINATION foldtrace RUNTIME DESTINATION foldtrace)
endif()
if(BUILD_TESTING)
  add_executable(test_continuation cpp/test_continuation.cpp)
  target_link_libraries(test_continuation PRIVATE continuation)
  add_test(NAME continuation_invariants COMMAND test_continuation)
endif()
