cmake_minimum_required(VERSION 3.20)
project(bucklemap VERSION 0.1.0 LANGUAGES CXX)
option(BUCKLEMAP_PYTHON "Build the Python extension" ON)
include(CTest)
add_library(beam INTERFACE)
target_include_directories(beam INTERFACE cpp)
target_compile_features(beam INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(beam INTERFACE /W4 /WX)
else()
  target_compile_options(beam INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUCKLEMAP_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/bindings.cpp)
  target_link_libraries(_core PRIVATE beam)
  install(TARGETS _core LIBRARY DESTINATION bucklemap RUNTIME DESTINATION bucklemap)
endif()
if(BUILD_TESTING)
  add_executable(test_beam cpp/test_beam.cpp)
  target_link_libraries(test_beam PRIVATE beam)
  add_test(NAME element_invariants COMMAND test_beam)
endif()
