cmake_minimum_required(VERSION 3.20)
project(yieldledger VERSION 0.1.0 LANGUAGES CXX)
option(YIELDLEDGER_PYTHON "Build Python extension" ON)
include(CTest)
add_library(material INTERFACE)
target_include_directories(material INTERFACE cpp)
target_compile_features(material INTERFACE cxx_std_17)
if(MSVC)
  target_compile_options(material INTERFACE /W4 /WX)
else()
  target_compile_options(material INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
if(YIELDLEDGER_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core cpp/bindings.cpp)
  target_link_libraries(_core PRIVATE material)
  install(TARGETS _core LIBRARY DESTINATION yieldledger RUNTIME DESTINATION yieldledger)
endif()
if(BUILD_TESTING)
  add_executable(test_material cpp/test_material.cpp)
  target_link_libraries(test_material PRIVATE material)
  add_test(NAME material_invariants COMMAND test_material)
endif()
