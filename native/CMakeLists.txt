cmake_minimum_required(VERSION 3.20)
project(thermomechanics_native LANGUAGES CXX)
include(CTest)
add_library(thermomechanics_native SHARED assembly.cpp)
target_compile_features(thermomechanics_native PUBLIC cxx_std_17)
target_compile_definitions(thermomechanics_native PRIVATE TM_BUILD)
target_include_directories(thermomechanics_native PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(thermomechanics_native PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(thermomechanics_native PRIVATE /W4 /WX)
else()
  target_compile_options(thermomechanics_native PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/library-$<CONFIG>.txt"
     CONTENT "$<TARGET_FILE:thermomechanics_native>\n")
if(BUILD_TESTING)
  add_executable(thermomechanics_invariants test_assembly.cpp)
  target_link_libraries(thermomechanics_invariants PRIVATE thermomechanics_native)
  add_test(NAME thermoelastic_native_invariants COMMAND thermomechanics_invariants)
endif()
